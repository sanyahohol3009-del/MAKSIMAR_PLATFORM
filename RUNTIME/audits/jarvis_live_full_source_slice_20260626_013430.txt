   320	def _apply_terminal_output_policy(response_text: str, *, grounded_answer: bool) -> tuple[str, bool, str]:
   321	    text = str(response_text or "")
   322	    limit = DEFAULT_GROUNDED_RESPONSE_MAX_CHARS if grounded_answer else DEFAULT_TERMINAL_RESPONSE_MAX_CHARS
   323	    if len(text) <= limit:
   324	        return text, False, ""
   325	    marker = "[output_truncated=true reason=terminal_response_size_cap next_action=ask_continue]"
   326	    trimmed = text[:limit].rstrip()
   327	    if not trimmed.endswith("\n"):
   328	        trimmed += "\n"
   329	    return trimmed + marker, True, "terminal_response_size_cap"

   407	def run_jarvis_live_brain_once(
   408	    user_text: str,
   409	    session_id: str = "default",
   410	    command_timeout_seconds: float | None = None,
   411	) -> dict[str, Any]:
   412	    chunks: list[str] = []
   413	    final_payload: dict[str, Any] = {}
   414	    selected_model_role = select_jarvis_live_model_role(user_text)
   415	    timeout_seconds = _command_timeout_seconds(
   416	        command_timeout_seconds,
   417	        str(selected_model_role.get("selected_model_role", "jarvis_chat_model")),
   418	    )
   419	    started_at = time.monotonic()
   420	    try:
   421	        for event in stream_jarvis_live_brain_response(
   422	            user_text,
   423	            session_id=session_id,
   424	            ollama_timeout_seconds=timeout_seconds,
   425	        ):
   426	            if time.monotonic() - started_at > timeout_seconds:
   427	                final_payload = _command_error_payload(user_text, "command_timeout")
   428	                break
   429	            if event["event"] == "chunk":
   430	                chunks.append(str(event["text"]))
   431	            if event["event"] == "done":
   432	                final_payload = dict(event)
   433	    except (TimeoutError, asyncio.CancelledError):
   434	        final_payload = _command_error_payload(user_text, "command_timeout_or_cancelled")
   435	    except KeyboardInterrupt:
   436	        final_payload = _command_error_payload(user_text, "command_cancelled_by_operator")
   437	    except Exception:
   438	        final_payload = _command_error_payload(user_text, "command_runtime_error")
   439	    if not final_payload:
   440	        final_payload = {
   441	            "event": "done",
   442	            "response_text": _sanitize_model_output("".join(chunks)).strip(),
   443	            "ollama_model_used": "",
   444	            "pc_control_allowed": False,
   445	            "output_truncated": False,
   446	            "output_truncation_reason": "",
   447	        }
   448	    response_text, output_truncated, output_truncation_reason = _apply_terminal_output_policy(
   449	        str(final_payload.get("response_text", "")).strip(),
   450	        grounded_answer=bool(final_payload.get("grounded_answer", False)),
   451	    )
   452	    if not response_text:
   453	        response_text = "Модельный runtime сейчас не вернул ответ. Повтори запрос или проверь Ollama статус."
   454	    return {
   455	        "llm_response": response_text,
   456	        "ollama_model_used": str(final_payload.get("ollama_model_used", "")),
   457	        "route_mode": str(final_payload.get("route_mode", "")),

   275	def _answer_project_command(text: str) -> str:
   276	    parts = text.split()
   277	    if len(parts) == 1:
   278	        return _format_project_snapshot_answer()
   279	    command = parts[1]
   280	    if command == "status":
   281	        return _format_project_status_answer()
   282	    if command == "tree":
   283	        return _format_tree_answer()
   284	    if command == "files":
   285	        page = _parse_int(parts[2], default=1) if len(parts) > 2 else 1
   286	        return _format_files_answer(page)
   287	    if command == "dirty":
   288	        return _format_dirty_answer()
   289	    if command == "search" and len(parts) > 2:
   290	        return _format_search_answer(" ".join(parts[2:]))
   291	    if command == "file" and len(parts) > 2:
   292	        page = _parse_int(parts[3], default=1) if len(parts) > 3 else 1
   293	        return _format_file_answer(parts[2], page, intent="PROJECT_FILE")
   294	    if command == "outline" and len(parts) > 2:
   295	        return _format_outline_answer(parts[2])
   296	    if command == "imports":
   297	        return _format_imports_answer(parts[2] if len(parts) > 2 else None)
   298	    if command == "tests":
   299	        return _format_tests_answer()
   300	    if command == "roadmap":
   301	        return _format_roadmap_answer()
   302	    if command == "models":
   303	        return _format_project_models_answer()
   304	    if command == "safety":
   305	        return _format_safety_answer()
   306	    return _format_project_help()

   397	def _format_project_semantic_search_answer(
   398	    user_text: str,
   399	    tool_route: dict[str, Any] | None = None,
   400	    semantic_query: str | None = None,
   401	    intent_label: str = "PROJECT_SEARCH",
   402	) -> tuple[str, int]:
   403	    route = tool_route or build_retrieval_readonly_tool_route("PROJECT_SEARCH", ("mgrep_readonly", "repo_search"), PROJECT_ROOT).to_read_model()
   404	    mgrep = inspect_mgrep_readonly_availability(PROJECT_ROOT).to_read_model()
   405	    query = semantic_query.strip() if semantic_query and semantic_query.strip() else _semantic_search_query(user_text)
   406	    payload = repo_search(query)
   407	    if not payload["results"] and query != user_text.strip():
   408	        payload = repo_search(user_text.strip())
   409	    path_hits = _project_path_matches(query)
   410	    if not payload["results"]:
   411	        if path_hits:
   412	            lines = [
   413	                f"[work] intent={intent_label}",
   414	                f"primary_tool={route['primary_tool']}",
   415	                f"effective_tool={route['effective_tool']}",
   416	                f"selected_tool_chain={_csv(route['selected_tool_chain'])}",
   417	                f"fallback_tool={route['fallback_tool']}",
   418	                f"fallback_reason={route['fallback_reason']}",
   419	                f"mgrep_source_present={str(mgrep['source_present']).lower()}",
   420	                f"mgrep_usable_now={str(mgrep['usable_now']).lower()}",
   421	                f"Нашёл по именам файлов read-only: query={query}; path_results={len(path_hits)}",
   422	            ]
   423	            lines.extend(f"- {path}" for path in path_hits[:20])
   424	            for path in path_hits[:2]:
   425	                outline = read_file_outline(path)
   426	                if outline.get("allowed"):
   427	                    lines.append(
   428	                        f"outline {path}: imports={_csv(outline['imports']) or 'none'}; "
   429	                        f"classes={_csv(outline['classes']) or 'none'}; "
   430	                        f"functions={_csv(outline['functions'][:12]) or 'none'}"
   431	                    )
   432	            lines.append("Read-only access confirmed; reads and analysis continue automatically.")
   433	            return "\n".join(lines), len(path_hits)
   434	        return (
   435	            f"[work] intent={intent_label}\n"
   436	            "Не нашёл подтверждение в текущих read-only источниках.\n"
   437	            f"primary_tool={route['primary_tool']}\n"
   438	            f"effective_tool={route['effective_tool']}\n"
   439	            f"selected_tool_chain={_csv(route['selected_tool_chain'])}\n"
   440	            f"fallback_tool={route['fallback_tool']}\n"
   441	            f"fallback_reason={route['fallback_reason']}\n"
   442	            f"mgrep_source_present={str(mgrep['source_present']).lower()}\n"
   443	            f"mgrep_usable_now={str(mgrep['usable_now']).lower()}\n"
   444	            "checked_tools=mgrep_readonly, repo_search, read_file_snippet, read_file_outline\n"
   445	            f"query={query}",
   446	            0,
   447	        )
   448	    lines = [
   449	        f"[work] intent={intent_label}",
   450	        f"primary_tool={route['primary_tool']}",
   451	        f"effective_tool={route['effective_tool']}",
   452	        f"selected_tool_chain={_csv(route['selected_tool_chain'])}",
   453	        f"fallback_tool={route['fallback_tool']}",
   454	        f"fallback_reason={route['fallback_reason']}",
   455	        f"mgrep_source_present={str(mgrep['source_present']).lower()}",
   456	        f"mgrep_usable_now={str(mgrep['usable_now']).lower()}",
   457	        f"Нашёл по проекту read-only: query={query}; results={payload['result_count']}",
   458	    ]
   459	    paths: list[str] = []
   460	    for result in payload["results"][:12]:
   461	        path = str(result["path"])
   462	        if path not in paths:
   463	            paths.append(path)
   464	        lines.append(f"- {path}:{result['line_number']}: {result['line']}")
   465	    for path in path_hits[:8]:
   466	        if path not in paths:
   467	            paths.append(path)
   468	            lines.append(f"- {path}: path_match")
   469	    for path in paths[:2]:
   470	        outline = read_file_outline(path)
   471	        if outline.get("allowed"):
   472	            lines.append(
   473	                f"outline {path}: imports={_csv(outline['imports']) or 'none'}; "
   474	                f"classes={_csv(outline['classes']) or 'none'}; "
   475	                f"functions={_csv(outline['functions'][:12]) or 'none'}"
   476	            )
   477	    lines.append("Read-only access confirmed; reads and analysis continue automatically.")
   478	    return "\n".join(lines), int(payload["result_count"]) + len(path_hits)
